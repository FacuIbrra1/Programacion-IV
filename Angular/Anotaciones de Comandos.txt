//Sirve para instalar ANGUNLAR, basicamente usar su consola:
npm install -g @angular/cli 

// Sirve para ver la version de Angular:
ng version 

// Sirve para crear un projecto:
ng new Clase-1

//Sirve para generar un componente:
 ng generate c componentes/miComponente

//Sirve para crear un servicio:
 ng g s servicios/miServicio


ANGULAR: COMANDOS ÚTILES
En la carpeta del proyecto: ng new nuevo-proyecto // crea un proyecto angular base

ng serve --port 4200 -o // corre la aplicación en el puerto 4200 y abre el navegador con la aplicación

ng g c 'components/mi-componente' // genero un componente en la ruta components/mi-componente

ng g s 'services/mi-servicio' // genero un servicio en la ruta services/ llamado mi-servicio

ng g m 'components/mi-componente' // genero un módulo dentro de components/mi-componente

ng g m app-routing --flat --module=app // genera un módulo de routing en la base del proyecto con el nombre app-routing

ng g c componentes/bienvenidos; ng g c componentes/error; ng g c componentes/login

ng g cl 'clases/mi-clase' // genero una clase en la ruta clases/ llamada mi-clase
